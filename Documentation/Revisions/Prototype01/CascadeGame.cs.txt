        public void StartLevel(int index)
        {
            if (State == Phase.Countdown) return;
            level = index; music.Stop(); pending = null; score = moves = 0; bestOccupied = balance.initialCount; journal.Clear();
            int seed = testSeed != 0 ? testSeed : unchecked((int)DateTime.UtcNow.Ticks);
            Model = new Board(balance, seed); rhythm = null; beatMap = null;
            music.clip = Resources.Load<AudioClip>("Audio/" + trackNames[level]);
            if (music.clip == null) { Error("Музыка недоступна. Нужен импортированный WAV."); return; }
            var mapAsset = Resources.Load<TextAsset>("Data/" + trackNames[level]);
            if (mapAsset != null) beatMap = JsonUtility.FromJson<BeatMap>(mapAsset.text);
            // Mechanic milestone intentionally has no invented beat grid or rhythm rewards.
            if (beatMap != null && beatMap.Valid(music.clip.length)) rhythm = new Rhythm(beatMap, balance);
            music.loop = false; music.pitch = 1; music.volume = volume; music.time = 0; music.clip.LoadAudioData();
            menu.SetActive(false); hud.SetActive(true); modal.SetActive(false); boardRoot.gameObject.SetActive(true);
            levelText.text = "УРОВЕНЬ " + (level + 1) + "\n" + trackNames[level];
            statusText.text = rhythm == null ? "ТЕСТ МЕХАНИКИ · РАЗМЕТКА В РАБОТЕ" : "ПОПАДАЙ В РИТМ · ОСВОБОЖДАЙ ПОЛЕ";
            countdown = 3; State = Phase.Countdown; pausedPosition = 0;
            Render(Model.cells); RefreshHud();
            UnityEngine.Debug.Log("Cascade seed=" + Model.seed + " config=" + balance.version + " track=" + trackNames[level]);
        }

        void Update()
        {
            if (safeRoot == null) return;
            Rect a = Screen.safeArea; safeRoot.anchorMin = a.min / new Vector2(Screen.width, Screen.height); safeRoot.anchorMax = a.max / new Vector2(Screen.width, Screen.height);
            float aspect = (float)Screen.width / Screen.height; gameCamera.orthographicSize = Mathf.Max(6.7f, 3.65f / aspect);
            if (State == Phase.Countdown)
            {
                if (music.clip.loadState == AudioDataLoadState.Failed) { Error("Не удалось декодировать трек."); return; }
                if (music.clip.loadState != AudioDataLoadState.Loaded) return;
                countdown -= Time.unscaledDeltaTime;
                statusText.text = countdown > 0 ? Mathf.CeilToInt(countdown).ToString() : "ПОЕХАЛИ";
                if (countdown <= 0)
                {
                    startDsp = AudioSettings.dspTime + .12; scheduledDsp = startDsp; music.PlayScheduled(startDsp);
                    eventAnchor = InputState.currentTime; dspAnchor = AudioSettings.dspTime; State = Phase.Playing;
                    statusText.text = rhythm == null ? "ТЕСТ МЕХАНИКИ · РАЗМЕТКА В РАБОТЕ" : "ЛОВИ РИТМ";
                }
            }
            if (State == Phase.Playing || State == Phase.Reaction)
            {
                rhythm?.Advance(Position);
                if (State == Phase.Reaction)
                {
                    reactionAge += Time.unscaledDeltaTime;
                    float duration = balance.presentationSeconds;
                    int stages = pending.waves.Count + 2;
                    int stage = Mathf.Min(stages - 1, Mathf.FloorToInt(reactionAge / duration * stages));
                    if (stage == 0) Render(pending.afterPlacement);
                    else if (stage <= pending.afterWaves.Count) Render(pending.afterWaves[stage - 1]);
                    else Render(pending.final);
                    if (reactionAge >= duration)
                    {
                        Render(pending.final);
                        if (pending.full) Finish(false); else if (Position >= music.clip.length) Finish(true); else State = Phase.Playing;
                        pending = null;
                    }
                }
                else if (Position >= music.clip.length) Finish(true);
                RefreshHud();
            }
            float pulse = rhythm == null || rhythm.Streak == 0 || beatMap.IsExcluded(Position) || reduced ? 0 : BeatPulse(Position);
            float strength = Mathf.Min(1, pulse * (rhythm == null ? 0 : .25f + rhythm.Multiplier * .2f));
            perimeter.startColor = perimeter.endColor = Color.Lerp(new Color(.55f,.14f,.65f), new Color(1,.65f,.25f), strength);
            foreach (var view in views)
            {
                view.border.startColor = view.border.endColor = Color.Lerp(new Color(.24f,.10f,.34f), new Color(.85f,.36f,1), strength);
                if (view.charge > 0 && State != Phase.Paused)
                {
                    float breathe = reduced ? 1 : 1 + .015f * Mathf.Sin(Time.unscaledTime * 2 + view.index);
                    view.orb.localScale = Vector3.one * .78f * breathe;
                }
            }
        }
        float BeatPulse(double t)
        {
            if (beatMap == null || beatMap.beats.Length == 0) return 0;
            int i = Array.BinarySearch(beatMap.beats, t); if (i < 0) i = ~i - 1;
            return i < 0 ? 0 : Mathf.Exp((float)(-(t - beatMap.beats[i]) * 12));
        }
        void InputEvent(InputEventPtr evt, InputDevice device)
        {
            if (!Application.isPlaying || (!evt.IsA<StateEvent>() && !evt.IsA<DeltaStateEvent>())) return;
            // The original Input System event time is mapped to the DSP clock, before Update ends a song.
            if (device is Touchscreen touch)
            {
                if (!touch.primaryTouch.press.ReadValueFromEvent(evt, out float press)) return;
                if (press <= .5f) { pointerTaken = false; return; }
                if (pointerTaken) return;
                if (!touch.primaryTouch.position.ReadValueFromEvent(evt, out Vector2 pos)) pos = touch.primaryTouch.position.ReadValue();
                pointerTaken = true; AcceptPointer(pos, evt.time);
            }
            else if (device is Mouse mouse)
            {
                if (!mouse.leftButton.ReadValueFromEvent(evt, out float press)) return;
                if (press <= .5f) { pointerTaken = false; return; }
                if (pointerTaken) return;
                if (!mouse.position.ReadValueFromEvent(evt, out Vector2 pos)) pos = mouse.position.ReadValue();
                pointerTaken = true; AcceptPointer(pos, evt.time);
            }
        }
        void AcceptPointer(Vector2 point, double eventTime)
        {
            if (State != Phase.Playing || AudioSettings.dspTime < scheduledDsp) return;
            double songTime = dspAnchor + eventTime - eventAnchor - startDsp;
            if (songTime < 0 || songTime >= music.clip.length) return;
            if (EventSystem.current != null)
            {
                var hits = new List<RaycastResult>(); EventSystem.current.RaycastAll(new PointerEventData(EventSystem.current) { position = point }, hits);
                if (hits.Count > 0) return;
            }
            if (!Physics.Raycast(gameCamera.ScreenPointToRay(point), out RaycastHit hit)) return;
            var cell = hit.collider.GetComponent<CascadeCell>(); if (cell == null) return;
            PlaceCell(cell.index, songTime);
        }
        public bool PlaceCell(int cell, double songTime)
        {
            if (State != Phase.Playing || songTime >= music.clip.length || songTime < 0) return false;
            if (cell < 0 || cell >= Model.cells.Length || Model.cells[cell] != 0)
            { journal.Add(new MoveLog { time = songTime, cell = cell, rejection = "occupied" }); return false; }
            var assessment = rhythm == null ? new RhythmResult { multiplier = 1, kind = "unmapped", error = double.NaN } : rhythm.Evaluate(songTime, offsetMs / 1000.0);
            pending = Model.Place(cell, assessment.multiplier); score += pending.points; moves++; bestOccupied = Math.Max(bestOccupied, Model.Occupied);
            journal.Add(new MoveLog { time = songTime, cell = cell, points = pending.points, waves = pending.waves.Count, occupied = Model.Occupied, streak = rhythm?.Streak ?? 0, rhythm = assessment.kind, error = double.IsNaN(assessment.error) ? 0 : assessment.error });
            statusText.text = rhythm == null ? "+" + pending.points + " · ТЕСТ МЕХАНИКИ" : assessment.kind == "hit" ? "В РИТМЕ  +" + pending.points : assessment.kind == "miss" ? "МИМО · СЕРИЯ СБРОШЕНА" : "+" + pending.points;
            reactionAge = 0; State = Phase.Reaction; Render(pending.afterPlacement); RefreshHud(); return true;
        }
        void Render(int[] values)
        {
            for (int i = 0; i < views.Length; i++)
            {
                var v = views[i]; int charge = Mathf.Clamp(values[i], 0, 4); v.charge = charge; v.orb.gameObject.SetActive(charge > 0);
                if (charge == 0) continue; v.shell.sharedMaterial = orbMaterials[charge - 1];
                for (int k = 0; k < v.symbols.Length; k++) v.symbols[k].SetActive(k == charge - 1);
            }
        }
        void RefreshHud()
        {
            scoreText.text = score.ToString("N0"); multiplierText.text = "×" + (rhythm?.Multiplier ?? 1);
            double t = Math.Min(Position, music.clip.length); if (State == Phase.Countdown) t = 0;
            timeText.text = Clock(t) + " / " + Clock(music.clip.length); progress.fillAmount = (float)(t / music.clip.length);
            
            nextText.text = "СЛЕДУЮЩАЯ"; nextText.color = Palette(Model.NextCharge);
            if (preview != null) { preview.orb.gameObject.SetActive(true); preview.shell.sharedMaterial = orbMaterials[Model.NextCharge-1]; for(int i=0;i<4;i++) preview.symbols[i].SetActive(i==Model.NextCharge-1); }
        }
        static string Clock(double t) => ((int)t / 60) + ":" + ((int)t % 60).ToString("00");
        public void Pause()
        {
            if (State != Phase.Playing && State != Phase.Reaction && State != Phase.Countdown) return;
            resumePhase = State; pausedPosition = State == Phase.Countdown ? 0 : Position;
            music.Pause(); State = Phase.Paused; modal.SetActive(true); modalTitle.text = "ПАУЗА";
            modalBody.text = "Музыка и поле ждут тебя"; SetModalButtons(true);
        }
        public void Resume()
        {
            if (State != Phase.Paused) return;
            modal.SetActive(false); State = resumePhase;
            if (State != Phase.Countdown)
            {
                startDsp = AudioSettings.dspTime - pausedPosition;
                if (pausedPosition <= 0) { startDsp = AudioSettings.dspTime + .12; scheduledDsp = startDsp; music.Stop(); music.PlayScheduled(startDsp); }
                else music.UnPause();
                eventAnchor = InputState.currentTime; dspAnchor = AudioSettings.dspTime;
            }
            pointerTaken = false;
        }
        void Finish(bool won)
        {
            if (State == Phase.Result) return;
            pausedPosition = Math.Min(Position, music.clip.length); State = Phase.Result; music.Stop(); modal.SetActive(true);
            if (won && rhythm != null) { int best = PlayerPrefs.GetInt("cascade.best." + level, 0); if (score > best) PlayerPrefs.SetInt("cascade.best." + level, score); PlayerPrefs.Save(); }
            modalTitle.text = won ? "ТРЕК ЗАВЕРШЁН" : "ПОЛЕ ЗАПОЛНЕНО";
            modalBody.text = score.ToString("N0") + " ОЧКОВ\n\nХодов: " + moves + "   •   Макс. занято: " + bestOccupied + "\nЛучшая серия: " + (rhythm?.Best ?? 0) + "\nПопаданий: " + (rhythm?.Hits ?? 0) + "   •   Промахов: " + (rhythm?.Misses ?? 0) + (rhythm == null ? "\n\nТест механики: ритм и рекорды отключены" : "");
            SetModalButtons(false); Export(won ? "completed" : "full");
        }
        void Export(string result)
        {
            var log = new AttemptLog { config = balance.version, track = trackNames[level], mapVersion = beatMap?.version ?? "unmapped", seed = Model.seed, score = score, result = result, moves = journal.ToArray() };
            try { File.WriteAllText(Path.Combine(Application.persistentDataPath, "cascade-last-attempt.json"), JsonUtility.ToJson(log, true)); }
            catch (Exception e) { UnityEngine.Debug.LogWarning(e.Message); }
        }

            modal.transform.Find("Settings").gameObject.SetActive(paused);
            modal.transform.Find("Effects").gameObject.SetActive(paused);
            modal.transform.Find("Cancel").gameObject.SetActive(false);
        }

            var btn = modal.transform.Find("Continue").GetComponent<Button>(); var label = btn.GetComponentInChildren<Text>();

        }

        {
            volume = volume >= .99f ? 0 : Mathf.Min(1,volume + .25f); music.volume = volume;
            PlayerPrefs.SetFloat("cascade.volume", volume); PlayerPrefs.SetInt("cascade.reduced", reduced ? 1 : 0); PlayerPrefs.Save();
            modalBody.text = "Громкость: " + Mathf.RoundToInt(volume * 100) + "%\nУменьшенные эффекты: " + (reduced ? "да" : "нет");
        }
        public void ToggleEffects()
        {
            reduced = !reduced; PlayerPrefs.SetInt("cascade.reduced",reduced ? 1 : 0); PlayerPrefs.Save();
            modalBody.text = "Уменьшенные эффекты: " + (reduced ? "да" : "нет");
        }

        Vector3[] Rounded(float width, float height, float radius, float z)
        {
            var points = new List<Vector3>();
            for (int corner = 0; corner < 4; corner++) for (int j = 0; j <= 8; j++)
            {
                float angle = (corner * 90 + j * 90f / 8) * Mathf.Deg2Rad;
                float cx = corner == 0 || corner == 3 ? width / 2 - radius : -width / 2 + radius;
                float cy = corner < 2 ? height / 2 - radius : -height / 2 + radius;
                points.Add(new Vector3(cx + radius * Mathf.Cos(angle),cy + radius * Mathf.Sin(angle),z));
            }
            return points.ToArray();
        }
